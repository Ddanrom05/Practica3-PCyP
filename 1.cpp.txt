#include <iostream>
#include <vector>
#include <string>
#include <thread>
#include <utility>

// Tarea: Comparar e intercambiar[cite: 2, 4]
void comparaSwap(std::vector<std::string>& A, int i, int j) {
    if (i < A.size() && j < A.size()) {
        if (A[i] > A[j]) {
            std::swap(A[i], A[j]);
        }
    }
}

void OrdenamientoImparPar(std::vector<std::string>& A) {
    int n = A.size();
    for (int fase = 0; fase < n; ++fase) { // N fases secuenciales[cite: 2]
        std::vector<std::thread> threads;

        if (fase % 2 == 0) {
            // Fase Par[cite: 2]
            for (int i = 0; i < n - 1; i += 2) {
                threads.push_back(std::thread(comparaSwap, std::ref(A), i, i + 1));
            }
        } else {
            // Fase Impar[cite: 2]
            for (int i = 1; i < n - 1; i += 2) {
                threads.push_back(std::thread(comparaSwap, std::ref(A), i, i + 1));
            }
        }

        // COEND: Esperar a que todos los hilos de la fase terminen[cite: 3, 4]
        for (auto& t : threads) {
            if (t.joinable()) {
                t.join();
            }
        }
    }
}

int main() {
    std::vector<std::string> A = {"piña", "manzana", "uva", "kiwi", "fresa"};

    std::cout << "Arreglo original:" << std::endl;
    for (const auto& str : A) {
        std::cout << str << " ";
    }
    std::cout << std::endl << std::endl;

    // Ordenamiento paralelo
    OrdenamientoImparPar(A);

    std::cout << "Arreglo ordenado:" << std::endl;
    for (const auto& str : A) {
        std::cout << str << " ";
    }
    std::cout << std::endl;

    return 0;
}
